// 1a
int ebenenen = 8;
for (int i = 1; i <= ebenenen; i++) {
    for (int j = 1; j <= i; j++) {
        System.out.print("*");
    }
    System.out.println();
}
// 1b
for (int i = 0; i <= ebenenen; i++) {
    for (int j = 1; j <= i; j++) {
        System.out.print("**");
    }
    System.out.println("*");
}
// 1c
int i = 1;
while (i <= ebenenen) {
    int j = 2;
    while (j <= i) {
        System.out.print("**");
        j++;
    }
    System.out.println("*");
    i++;
}
